import { useEffect, useState } from 'react';
import { Truck, Wrench, CheckCircle2 } from 'lucide-react';

// Composant Loader principal
export default function AutomotiveLoader() {
  const [progress, setProgress] = useState(0);
  const [currentStep, setCurrentStep] = useState(0);

  const loadingSteps = [
    { icon: Truck, text: "Chargement de la flotte...", color: "text-green-500" },
    { icon: Wrench, text: "Initialisation du système...", color: "text-blue-500" },
    { icon: CheckCircle2, text: "Prêt !", color: "text-emerald-500" }
  ];

  useEffect(() => {
    const interval = setInterval(() => {
      setProgress(prev => {
        if (prev >= 100) {
          clearInterval(interval);
          return 100;
        }
        return prev + 1;
      });
    }, 30);

    return () => clearInterval(interval);
  }, []);

  useEffect(() => {
    if (progress < 40) setCurrentStep(0);
    else if (progress < 80) setCurrentStep(1);
    else setCurrentStep(2);
  }, [progress]);

  const CurrentIcon = loadingSteps[currentStep].icon;

  return (
    <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-800 to-green-900 flex items-center justify-center p-4">
      <div className="max-w-md w-full space-y-8">
        {/* Logo */}
        <div className="text-center">
          <div className="inline-flex items-center justify-center w-20 h-20 bg-green-500/10 rounded-2xl mb-4 backdrop-blur-sm border border-green-500/20">
            <Truck className="w-10 h-10 text-green-500" />
          </div>
          <h1 className="text-3xl font-bold text-white mb-2">Yaro FleetOS</h1>
          <p className="text-slate-400">CMMS & Fleet Management</p>
        </div>

        {/* Camion animé */}
        <div className="relative h-24 bg-slate-800/50 rounded-xl overflow-hidden border border-slate-700/50 backdrop-blur-sm">
          {/* Route */}
          <div className="absolute bottom-0 left-0 right-0 h-2 bg-slate-700">
            <div className="h-full flex items-center justify-around">
              <div className="w-8 h-0.5 bg-yellow-500/50"></div>
              <div className="w-8 h-0.5 bg-yellow-500/50"></div>
              <div className="w-8 h-0.5 bg-yellow-500/50"></div>
              <div className="w-8 h-0.5 bg-yellow-500/50"></div>
            </div>
          </div>

          {/* Camion en mouvement */}
          <div 
            className="absolute bottom-4 transition-all duration-300 ease-out"
            style={{ left: `${progress}%`, transform: 'translateX(-50%)' }}
          >
            <div className="relative">
              {/* Corps du camion */}
              <div className="flex items-end">
                {/* Cabine */}
                <div className="w-8 h-10 bg-gradient-to-br from-green-500 to-green-600 rounded-tl-lg relative">
                  {/* Fenêtre */}
                  <div className="absolute top-1 left-1 right-1 h-3 bg-green-300/30 rounded-tl-lg"></div>
                </div>
                {/* Remorque */}
                <div className="w-16 h-8 bg-gradient-to-br from-green-600 to-green-700 rounded-tr-lg"></div>
              </div>

              {/* Roues */}
              <div className="absolute -bottom-2 left-1 flex gap-10">
                <div className="w-4 h-4 bg-slate-900 rounded-full border-2 border-slate-600 animate-spin" 
                     style={{ animationDuration: '0.5s' }}>
                  <div className="w-1 h-1 bg-slate-400 rounded-full absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2"></div>
                </div>
                <div className="w-4 h-4 bg-slate-900 rounded-full border-2 border-slate-600 animate-spin" 
                     style={{ animationDuration: '0.5s' }}>
                  <div className="w-1 h-1 bg-slate-400 rounded-full absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2"></div>
                </div>
              </div>

              {/* Fumée d'échappement */}
              {progress > 5 && (
                <div className="absolute -left-2 top-2 flex gap-1">
                  <div className="w-2 h-2 bg-slate-500/40 rounded-full animate-ping" 
                       style={{ animationDuration: '1s' }}></div>
                </div>
              )}
            </div>
          </div>

          {/* Paysage décoratif (nuages/arbres qui défilent) */}
          <div className="absolute inset-0 pointer-events-none opacity-20">
            <div className="absolute top-2 animate-pulse" 
                 style={{ 
                   left: `${20 - progress * 0.3}%`,
                   animationDuration: '2s'
                 }}>
              <div className="w-6 h-6 bg-slate-600/30 rounded-full"></div>
            </div>
            <div className="absolute top-4 animate-pulse" 
                 style={{ 
                   left: `${60 - progress * 0.5}%`,
                   animationDuration: '3s',
                   animationDelay: '0.5s'
                 }}>
              <div className="w-8 h-8 bg-slate-600/30 rounded-full"></div>
            </div>
          </div>
        </div>

        {/* Barre de progression */}
        <div className="space-y-3">
          <div className="h-3 bg-slate-800/50 rounded-full overflow-hidden border border-slate-700/50 backdrop-blur-sm">
            <div 
              className="h-full bg-gradient-to-r from-green-500 via-green-400 to-emerald-500 transition-all duration-300 ease-out relative overflow-hidden"
              style={{ width: `${progress}%` }}
            >
              {/* Effet de brillance */}
              <div className="absolute inset-0 bg-gradient-to-r from-transparent via-white/20 to-transparent animate-shimmer"
                   style={{ 
                     backgroundSize: '200% 100%',
                     animation: 'shimmer 2s infinite'
                   }}></div>
            </div>
          </div>

          {/* Texte de progression */}
          <div className="flex items-center justify-between text-sm">
            <div className="flex items-center gap-2">
              <CurrentIcon className={`w-5 h-5 ${loadingSteps[currentStep].color} animate-pulse`} />
              <span className="text-slate-300">{loadingSteps[currentStep].text}</span>
            </div>
            <span className="text-green-400 font-mono font-semibold">{progress}%</span>
          </div>
        </div>

        {/* Indicateurs de statut */}
        <div className="flex justify-center gap-2">
          {loadingSteps.map((step, index) => (
            <div
              key={index}
              className={`w-2 h-2 rounded-full transition-all duration-300 ${
                index <= currentStep 
                  ? 'bg-green-500 scale-100' 
                  : 'bg-slate-700 scale-75'
              }`}
            ></div>
          ))}
        </div>
      </div>

      <style>{`
        @keyframes shimmer {
          0% { background-position: -200% 0; }
          100% { background-position: 200% 0; }
        }
      `}</style>
    </div>
  );
}

// Composant d'intégration pour votre Landing Page
export function LoaderWrapper({ children, loading }: { children: React.ReactNode, loading: boolean }) {
  if (loading) {
    return <AutomotiveLoader />;
  }
  return <>{children}</>;
}
