4. Bonus : remplir les jours manquants “à la veille”
Pour respecter la règle « si un jour manque, on reprend la veille », génère la série côté SQL :

sql
Copier
Modifier
WITH d AS (
  SELECT generate_series(
           ${startDate}::date,
           ${endDate}::date,
           '1 day'
         ) AS d
),
latest AS (
  SELECT v.id,
         (SELECT sh.status
            FROM status_history sh
           WHERE sh.vehicle_id = v.id
             AND sh.date <= d.d
           ORDER BY sh.date DESC
           LIMIT 1) AS status
    FROM vehicles v, d
   WHERE v.created_at::date <= d.d
)
SELECT d.d AS date,
       COUNT(*) FILTER (WHERE status = 'operational') AS operational,
       COUNT(*) FILTER (WHERE status = 'maintenance')  AS maintenance,
       COUNT(*) FILTER (WHERE status = 'out_of_service') AS out_of_service
  FROM d
  JOIN latest ON TRUE
 GROUP BY d.d
 ORDER BY d.d;
